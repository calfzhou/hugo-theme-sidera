"notebook" -}} {{- $owner = strings.TrimPrefix "content/" $dir -}} {{- if not (findRE `^[a-z0-9-]+(/[a-z0-9-]+)*$` $owner) -}} {{- errorf "P1B notebook content path must use lowercase ASCII slug $logical -}}{{- end -}} {{- end -}} {{- if in (slice "tags" "tags.md" "page" "page.md") $name -}} {{- errorf "P1B reserved notebook source namespace: %s" (path.Join $dir .Name) -}} {{- end -}} {{- end {{- $nodes.SetInMap "routes" (printf "%s/tags" $owner) (dict "key" "" "slug" "") -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- $leaf := $local.leaf -}} {{- range os.ReadDir $dir -}} {{- range $name, $physical := $local.files -}} {{- $file := path.Join $dir $physical -}} {{- if and $owner (ne $name "_index.md") (or (not $leaf) (eq $name "index.md")) -}} {{- $fm := partial {{- $slug := delimit (first . $slugs) "/" -}} {{- $route := printf "%s/tags/%s" $owner $slug -}} {{- with index ($nodes.Get "routes") $route -}} {{- if ne .key $key -}} {{- errorf "P1B tag slug collision at %s: %q and %q (%s)" $route .key $key $file -}} {{- end -}} {{- end -}} {{- $nodes.SetInMap "routes" $route (dict "key" $key "slug" $slug) -}} {{- end -}} {{- end -}} {{- end -}} {{- end
