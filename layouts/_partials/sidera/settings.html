{{- /* Native Page.Params first; own selected-section defaults, then ancestor descendant defaults. Ordinary ancestor Params never implicitly cascade. */ -}} {{- $page := . -}}{{- $owner := partial "collection-owner.html" . -}} {{- $defaults := partial "model/defaults.html" . -}} {{- $scopes := slice $page.Params -}} {{- if and (or $page.Params.sidera.tag_view $page.Params.sidera.archive_view) $owner -}} {{- /* A contextual navigation Page queries its owner policy, not article metadata. */ -}} {{- $scopes = $scopes | append (partialCached "sidera/settings.html" $owner $owner.Language.Lang $owner.Path) -}} {{- else -}} {{- $stop := false -}} {{- if $page.IsSection -}} {{- with partial "model/preset.html" $page -}}{{- with .Params.defaults.params -}}{{- $scopes = $scopes | append . -}}{{- end -}}{{- end -}} {{- if and (isset $page.Params "preset") (eq (len ($page.GetTerms "preset")) 0) -}}{{- $stop = true -}}{{- end -}} {{- end -}} {{- range $page.Ancestors -}} {{- if $stop -}}{{- break -}}{{- end -}} {{- if and .IsSection (not .Params.sidera.tag_view) -}} {{- with partial "model/preset.html" . -}}{{- with .Params.defaults.cascade.params -}}{{- $scopes = $scopes | append . -}}{{- end -}}{{- end -}} {{- if and (isset .Params "preset") (eq (len (.GetTerms "preset")) 0) -}}{{- $stop = true -}}{{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- $scopes = $scopes | append $page.Site.Params | append $defaults -}} {{- $settings := dict -}} {{- range $key, $_ := $defaults -}} {{- range $scopes -}}{{- if isset . $key -}}{{- $settings = merge $settings (dict $key (index . $key)) -}}{{- break -}}{{- end -}}{{- end -}} {{- end -}} {{- $identity := dict -}}{{- if isset $page.Site.Params "identity" -}}{{- $identity = $page.Site.Params.identity -}}{{- end -}} {{- $settings = merge $settings (dict "identity" $identity) -}} {{- return (partial "model/settings-valid.html" (dict "page" $page "settings" $settings)) -}}
